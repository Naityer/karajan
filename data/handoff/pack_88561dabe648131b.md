### `scripts/report-skills.ps1`

```
param(
  [string]$AtlasRoot = (Resolve-Path (Join-Path $PSScriptRoot '..\..\atlas'))
)

$ErrorActionPreference = 'Stop'
$today = Get-Date -Format 'yyyy-MM-dd'
$skillRoots = @(
  @{ Name = 'Codex'; Path = Join-Path $env:USERPROFILE '.codex\skills' },
  @{ Name = 'Agentes'; Path = Join-Path $env:USERPROFILE '.agents\skills' }
)

$skills = @{}
foreach ($root in $skillRoots) {
  if (-not (Test-Path $root.Path)) { continue }
  foreach ($file in Get-ChildItem $root.Path -Recurse -Filter SKILL.md -File -ErrorAction SilentlyContinue) {
    $raw = Get-Content -Raw $file.FullName
    $name = if ($raw -match '(?m)^name:\s*(.+?)\s*$') { $matches[1].Trim('"'' ') } else { Split-Path $file.DirectoryName -Leaf }
    if (-not $name) { continue }
    $description = if ($raw -match '(?m)^description:\s*(.+?)\s*$') { $matches[1].Trim('"'' ') } else { '' }
    if (-not $description -or $description -in @('>', '|', '/-', '/')) {
      $description = ($raw -split "`r?`n" | Where-Object {
        $_.Trim() -and $_ -notmatch '^(---|#|name:|description:|origin:)'
      } | Select-Object -First 1).Trim()
    }
    $key = $name.ToLowerInvariant()
    if (-not $skills.ContainsKey($key)) {
      $skills[$key] = [pscustomobject]@{
        Name = $name
        Description = ($description -replace '\|', '\|' -replace '\s+', ' ').Trim()
        Locations = @()
      }
    }
    $skills[$key].Locations += "$($root.Name): ``$($file.FullName)``"
  }
}

$componentRows = @(
  '| Dify | Instalado (checkout) | `2c6ec1a` | Sí (`f68cabe`) | Workflows visuales, RAG, agentes y observabilidad. |',
  '| LlamaIndex | Instalado (Python) | `0.14.23` | Consultar PyPI al actualizar | Ingesta, índices, recuperación, RAG y OCR. |',
  '| Browser-use | Instalado (Python) | `0.13.3` | Consultar PyPI al actualizar | Automatización de navegación e interacción web. |',
  '| ECC | Instalado | skills locales | Gestionado por su sincronizador | Reglas, memoria, seguridad y flujo de ingeniería. |',
  '| gstack | Instalado (checkout) | `11de390` | Sí (`a325940`) | Roles de producto, diseño, QA, seguridad y release. |',
  '| Bun | Instalado | `1.3.14` | Consultar WinGet | Runtime y gestor JavaScript requerido por gstack. |',
  '| Ponytail | Codex disponible | `codex-cli 0.143.0` | Marketplace | Disciplina YAGNI y mínimo código seguro. |',
  '| Taste Skill | Instalado (checkout) | `b177427` | Sí (`e988add`) | Criterio creativo y de producto. |',
  '| Impeccable | Instalado (checkout) | `f40e2f8` | Sí (`e0144ed`) | Precisión, acabado y revisión de interfaces. |',
  '| Emil Kowalski Skills | Instalado (checkout) | `f76bece` | Sí (`70744e3`) | Patrones de diseño, componentes y animación. |',
  '| Anthropic Skills | Instalado (checkout) | `9d2f1ae` | Sí (`b29e7cf`) | Documentos, PDF, ofimática y utilidades oficiales. |',
  '| Scientific Agent Skills | Instalado (checkout) | `4d97e29` | Sí (`81e316b`) | Investigación, análisis, literatura, laboratorio y visualización. |',
  '| Graphify | Instalado (checkout) | `9c27a52` | Sí (`1644230`) | Grafo de conocimiento del repositorio. |',
  '| DSPy | Instalado (Python) | `3.2.1` | Consultar PyPI al actualizar | Pipelines LLM declarativos y optimizables. |',
  '| Aider | Instalado (CLI) | `0.86.2` | Consultar PyPI al actualizar | Pair programming multi-modelo en terminal. |',
  '| Headroom | Instalado (CLI) | `0.30.0` | Consultar PyPI al actualizar | Compresión de contexto para reducir tokens. |',
  '| Promptfoo | Instalado (CLI) | `0.121.17` | Sí (`0.121.19`) | Evals, aserciones y regresión de prompts/modelos. |'
)

$lines = @(
  '---',
  'id: naityer_setup.complementos-ia',
  'app: naityer_setup',
  'title: Complementos IA y skills disponibles',
  'responsibility: inventario de capacidades',
  'kind: spec',
  'tags: [skills, complementos, estado, actualizaciones]',
  'status: published',
  "updated: $today",
  '---',
  '',
  '# Complementos IA y skills disponibles',
  '',
  "> Inventario generado el $today desde ``~/.codex/skills`` y ``~/.agents/skills``. Las comprobaciones remotas se hicieron por colección; una skill individual hereda el estado de su colección.",
  '',
  '## Estado de los componentes',
  '',
  '| Componente | Estado | Versión/commit local | Actualización | Para qué sirve |',
  '| --- | --- | --- | --- | --- |'
) + $componentRows + @(
  '',
  "## Skills instaladas ($($skills.Count) únicas)",
  '',
  '| Skill | Para qué sirve | Estado y ubicación |',
  '| --- | --- | --- |'
)

foreach ($skill in $skills.Values | Sort-Object Name) {
  $purpose = if ($skill.Description) { $skill.Description } else { 'Sin descripción estructurada en su SKILL.md.' }
  $locations = ($skill.Locations | Sort-Object -Unique) -join '<br>'
  $lines += "| ``$($skill.Name)`` | $purpose | Instalada · $locations |"
}

$lines += @(
  '',
  '## Criterio de actualización',
  '',
  '- `Sí`: el HEAD remoto de la colección no coincide con el checkout local.',
  '- `Gestionado por su sincronizador`: no existe un upstream por skill; se actualiza mediante el instalador del componente.',
  '- Las versiones PyPI/WinGet deben verificarse de nuevo justo antes de actualizar, porque son datos cambiantes.',
  '- No se actualizó ningún componente durante este inventario.',
  ''
)

$text = $lines -join "`n"
$repoRoot = Resolve-Path (Join-Path $PSScriptRoot '..')
$source = Join-Path $repoRoot 'docs\complementos-ia.md'
$mirror = Join-Path $AtlasRoot 'docs\apps\naityer_setup\complementos-ia.md'
New-Item -ItemType Directory -Force (Split-Path $source -Parent), (Split-Path $mirror -Parent) | Out-Null
Set-Content -Path $source -Value $text -Encoding utf8
Set-Content -Path $mirror -Value $text -Encoding utf8
Write-Output "Generadas $($skills.Count) skills: $source"
Write-Output "Espejo Atlas dev: $mirror"
```
